<objective>
Rewrite http.ts to the simplified MCP-only Express server shape, then update test-phase2.sh to verify the new behavior and document the AUTH-02 token refresh procedure.

Script template:
```bash
#!/usr/bin/env bash
# Phase 2 automated verification — run from tools/hlx-admin-mcp/
# Usage: bash test-phase2.sh
# Requirements tested:
#   AUTH-01: HLX_ADMIN_TOKEN env var wired in tools.ts (static check)
#   AUTH-02: Token refresh procedure documented below (procedural — no automated check)
#   HLX-01 through HLX-08: hlx_* tools present in TOOLS array (smoke: tools/list)
#
# AUTH-02 Token Refresh Procedure:
#   Option A (preferred — 1-year key):
#     1. Temporarily set a valid IMS token as HLX_ADMIN_TOKEN on Railway
#     2. Call hlx_create_apikey to generate a 1-year API key for the site
#     3. Replace HLX_ADMIN_TOKEN with the new API key value
#     4. Redeploy — key lasts ~1 year, no demo-day refresh needed
#   Option B (short-lived IMS token — ~24h):
#     1. Morning of demo: obtain fresh IMS token from aem.live or Adobe Developer Console
#     2. Update HLX_ADMIN_TOKEN on Railway (Settings → Variables)
#     3. Railway auto-redeploys with the new token

set -e
PASS=0
FAIL=0
SERVER_PID=""

pass() { echo "[PASS] $1"; PASS=$((PASS+1)); }
fail() { echo "[FAIL] $1"; FAIL=$((FAIL+1)); }

cleanup() {
  if [ -n "$SERVER_PID" ]; then
    kill "$SERVER_PID" 2>/dev/null || true
    wait "$SERVER_PID" 2>/dev/null || true
    SERVER_PID=""
  fi
  rm -f /tmp/phase2-body.tmp
}
trap cleanup EXIT

echo "=== Phase 2 Static Checks ==="

# AUTH-01: HLX_ADMIN_TOKEN is referenced in tools.ts
if grep -q 'HLX_ADMIN_TOKEN' src/tools.ts 2>/dev/null; then
  pass "AUTH-01: HLX_ADMIN_TOKEN referenced in tools.ts"
else
  fail "AUTH-01: HLX_ADMIN_TOKEN not found in tools.ts"
fi

# AUTH-01: No OAuth routes in http.ts
if ! grep -qE 'app\.(get|post)\(.*/(login|callback)' src/http.ts 2>/dev/null; then
  pass "AUTH-01: No /login or /callback routes in http.ts"
else
  fail "AUTH-01: OAuth routes still present in http.ts"
fi

# No DA auth tools in TOOLS array
if ! grep -qE '"(da_login|da_logout|da_whoami)"' src/tools.ts 2>/dev/null; then
  pass "DA tools: da_login/da_logout/da_whoami removed from tools.ts"
else
  fail "DA tools: da_login/da_logout/da_whoami still present in tools.ts"
fi

echo ""
echo "=== Phase 2 Smoke Checks (starting server on PORT=9999) ==="

npm run build 2>/dev/null || true

PORT=9999 node dist/http.js &
SERVER_PID=$!
sleep 2

# Health check
HTTP_CODE=$(curl -s -o /dev/null -w "%{http_code}" http://localhost:9999/health 2>/dev/null || echo "000")
if [ "$HTTP_CODE" = "200" ]; then
  pass "Health: GET /health returns 200"
else
  fail "Health: GET /health returned $HTTP_CODE (expected 200)"
fi

# MCP initialize
HTTP_CODE=$(curl -s -o /dev/null -w "%{http_code}" \
  -X POST http://localhost:9999/mcp \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":"2024-11-05","capabilities":{},"clientInfo":{"name":"test","version":"0"}}}' \
  2>/dev/null || echo "000")
if [ "$HTTP_CODE" = "200" ]; then
  pass "MCP: POST /mcp initialize returns 200"
else
  fail "MCP: POST /mcp initialize returned $HTTP_CODE (expected 200)"
fi

# MCP tools/list — confirm hlx_status and hlx_preview are present
curl -s -o /tmp/phase2-body.tmp \
  -X POST http://localhost:9999/mcp \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":2,"method":"tools/list"}' \
  2>/dev/null || true
if grep -q '"hlx_status"' /tmp/phase2-body.tmp && grep -q '"hlx_preview"' /tmp/phase2-body.tmp; then
  pass "HLX tools: tools/list contains hlx_status and hlx_preview"
else
  fail "HLX tools: tools/list missing hlx_status or hlx_preview (response: $(cat /tmp/phase2-body.tmp 2>/dev/null | head -c 200))"
fi

# tools/list must NOT include da_login
if ! grep -q '"da_login"' /tmp/phase2-body.tmp 2>/dev/null; then
  pass "DA auth tools: da_login absent from tools/list"
else
  fail "DA auth tools: da_login still present in tools/list"
fi

# MCP returns 200 even without auth (no 401 gate)
HTTP_CODE=$(curl -s -o /dev/null -w "%{http_code}" \
  -X POST http://localhost:9999/mcp \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":3,"method":"tools/list"}' \
  2>/dev/null || echo "000")
if [ "$HTTP_CODE" = "200" ]; then
  pass "Auth: POST /mcp without Bearer returns 200 (no 401 gate)"
else
  fail "Auth: POST /mcp returned $HTTP_CODE without Bearer (expected 200, not 401)"
fi

# Invalid JSON-RPC returns 400
HTTP_CODE=$(curl -s -o /dev/null -w "%{http_code}" \
  -X POST http://localhost:9999/mcp \
  -H "Content-Type: application/json" \
  -d '{"not":"jsonrpc"}' \
  2>/dev/null || echo "000")
if [ "$HTTP_CODE" = "400" ]; then
  pass "Validation: malformed JSON-RPC returns 400"
else
  fail "Validation: malformed JSON-RPC returned $HTTP_CODE (expected 400)"
fi

cleanup

echo ""
echo "=== Results: $PASS passed, $FAIL failed ==="

if [ "$FAIL" -gt 0 ]; then
  exit 1
fi
```
